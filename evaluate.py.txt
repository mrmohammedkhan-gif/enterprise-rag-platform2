from rag import generate_answer

test_cases = [
    {
        "question": "What is required for privileged access?",
        "expected_source": "security-policy.txt"
    },
    {
        "question": "What must AI systems be evaluated for?",
        "expected_source": "ai-governance-policy.txt"
    },
    {
        "question": "Who must use multi-factor authentication?",
        "expected_source": "security-policy.txt"
    }
]

passed = 0

for test in test_cases:
    answer, docs = generate_answer(test["question"])

    retrieved_sources = [doc["source"] for doc in docs]
    success = test["expected_source"] in retrieved_sources

    if success:
        passed += 1

    print("=" * 60)
    print("Question:", test["question"])
    print("Expected source:", test["expected_source"])
    print("Retrieved sources:", retrieved_sources)
    print("Passed:", success)
    print("Answer:", answer)

print("=" * 60)
print(f"Retrieval score: {passed}/{len(test_cases)}")
